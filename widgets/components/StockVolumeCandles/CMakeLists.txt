cmake_minimum_required(VERSION 3.20)
project(StockVolumeCandles LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockVolumeCandles main_StockVolumeCandles.cpp)
target_compile_options(StockVolumeCandles PRIVATE -Wall -Wextra)
target_link_libraries(StockVolumeCandles PRIVATE X11::X11)
