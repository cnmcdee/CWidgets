cmake_minimum_required(VERSION 3.20)
project(StockAccumulationDistribution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockAccumulationDistribution main_StockAccumulationDistribution.cpp)
target_compile_options(StockAccumulationDistribution PRIVATE -Wall -Wextra)
target_link_libraries(StockAccumulationDistribution PRIVATE X11::X11)
