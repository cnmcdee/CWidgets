cmake_minimum_required(VERSION 3.20)
project(cwidgets_layout LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(cwidgets_layout main_layout.cpp)
target_include_directories(cwidgets_layout PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(cwidgets_layout PRIVATE X11::X11)
