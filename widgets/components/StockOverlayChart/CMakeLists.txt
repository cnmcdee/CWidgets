cmake_minimum_required(VERSION 3.20)
project(StockOverlayChart LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockOverlayChart main_StockOverlayChart.cpp)
target_compile_options(StockOverlayChart PRIVATE -Wall -Wextra)
target_link_libraries(StockOverlayChart PRIVATE X11::X11)
