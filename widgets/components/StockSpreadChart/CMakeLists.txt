cmake_minimum_required(VERSION 3.20)
project(StockSpreadChart LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockSpreadChart main_StockSpreadChart.cpp)
target_compile_options(StockSpreadChart PRIVATE -Wall -Wextra)
target_link_libraries(StockSpreadChart PRIVATE X11::X11)
