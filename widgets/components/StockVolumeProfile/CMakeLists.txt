cmake_minimum_required(VERSION 3.20)
project(StockVolumeProfile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockVolumeProfile main_StockVolumeProfile.cpp)
target_compile_options(StockVolumeProfile PRIVATE -Wall -Wextra)
target_link_libraries(StockVolumeProfile PRIVATE X11::X11)
