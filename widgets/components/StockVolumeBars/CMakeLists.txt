cmake_minimum_required(VERSION 3.20)
project(StockVolumeBars LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockVolumeBars main_StockVolumeBars.cpp)
target_compile_options(StockVolumeBars PRIVATE -Wall -Wextra)
target_link_libraries(StockVolumeBars PRIVATE X11::X11)
