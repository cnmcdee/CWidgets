cmake_minimum_required(VERSION 3.20)
project(StockOBVChart LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(StockOBVChart main_StockOBVChart.cpp)
target_compile_options(StockOBVChart PRIVATE -Wall -Wextra)
target_link_libraries(StockOBVChart PRIVATE X11::X11)
